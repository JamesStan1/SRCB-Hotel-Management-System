import { NextResponse } from 'next/server';
import pool from '../../lib/db'; // Adjust the import path as necessary

export async function GET() {
	let client;
	try {
		client = await pool.connect();
		const result = await client.query(`
			SELECT 
				a.id, a.user_id, a.date, a.clock_in, a.clock_out, a.hours_worked, a.status,
				u.name AS user_name, u.role AS user_role, u.email
			FROM attendance a
			JOIN users u ON a.user_id = u.id
			ORDER BY a.date DESC, u.name ASC
		`);
		return NextResponse.json(result.rows, { status: 200 });
	} catch (error) {
		console.error('Error fetching attendance:', error.message, error.code, error.stack);
		return NextResponse.json({ error: 'Failed to fetch attendance', details: error.message }, { status: 500 });
	} finally {
		if (client) client.release();
	}
}

export async function POST(request) {
	let client;
	try {
		const { userId, action, date } = await request.json();
		if (!userId || !action || !date) {
			return NextResponse.json({ error: 'User ID, action, and date are required' }, { status: 400 });
		}
		if (!['clock_in', 'clock_out'].includes(action)) {
			return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
		}

		client = await pool.connect();

		// Check user status
		const userRes = await client.query(`
			SELECT status FROM users WHERE id = $1
		`, [userId]);
		if (userRes.rows.length === 0) {
			return NextResponse.json({ error: 'User not found' }, { status: 404 });
		}
		if (userRes.rows[0].status === 'Paused') {
			return NextResponse.json({ error: 'User is paused, cannot record attendance' }, { status: 400 });
		}

		if (action === 'clock_in') {
			// Check if already clocked in
			const existing = await client.query(`
				SELECT id FROM attendance WHERE user_id = $1 AND date = $2
			`, [userId, date]);
			if (existing.rows.length > 0) {
				return NextResponse.json({ error: 'Already clocked in for this date' }, { status: 400 });
			}

			// Record clock-in
			await client.query(`
				INSERT INTO attendance (user_id, date, clock_in, status)
				VALUES ($1, $2, CURRENT_TIME, 'present')
			`, [userId, date]);
			return NextResponse.json({ message: 'Clocked in successfully' }, { status: 200 });
		} else {
			// Update clock-out
			const result = await client.query(`
				UPDATE attendance 
				SET clock_out = CURRENT_TIME, status = 'present'
				WHERE user_id = $1 AND date = $2 AND clock_in IS NOT NULL AND clock_out IS NULL
				RETURNING id
			`, [userId, date]);
			if (result.rows.length === 0) {
				return NextResponse.json({ error: 'No active clock-in found for this date' }, { status: 400 });
			}
			return NextResponse.json({ message: 'Clocked out successfully' }, { status: 200 });
		}
	} catch (error) {
		console.error('Attendance error:', error.message, error.code, error.stack);
		return NextResponse.json({ error: 'Failed to process attendance', details: error.message }, { status: 500 });
	} finally {
		if (client) client.release();
	}
}

import { NextResponse } from 'next/server';
import pool from '../../../lib/db'; // Adjust the import path as necessary

export async function POST(request) {
	let client;
	try {
		const { userId, clockInTime } = await request.json();
		if (!userId) {
			return NextResponse.json({ error: 'User ID is required' }, { status: 400 });
		}

		console.log('Attempting to connect to database...');
		client = await pool.connect().catch(err => {
			console.error('Connection error:', err.message, err.code, err.stack);
			throw new Error(`Failed to connect to database: ${err.message}`);
		});

	// Use local date (YYYY-MM-DD) to match client-side date selection
	// so that when users in different timezones clock in near midnight, the
	// recorded date matches the client's selected date.
	const today = new Date().toLocaleDateString('en-CA');

		// Check if already clocked in today
		const existing = await client.query(`
			SELECT clock_in, clock_out FROM attendance
			WHERE user_id = $1 AND date = $2
		`, [userId, today]);

		if (existing.rows.length > 0) {
			const record = existing.rows[0];
			if (record.clock_in) {
				return NextResponse.json({ 
					error: 'Already clocked in today',
					clockInTime: record.clock_in 
				}, { status: 400 });
			}
		}

		// Insert or update and return the recorded clock_in time so client can
		// display the accurate clock-in time immediately. If the client provided
		// a local time string (HH:MM:SS) use that to avoid server/DB timezone
		// differences; otherwise fall back to CURRENT_TIME.
		let res;
		if (clockInTime) {
			res = await client.query(`
			INSERT INTO attendance (user_id, date, clock_in)
			VALUES ($1, $2, $3::time)
			ON CONFLICT (user_id, date) DO UPDATE
			SET clock_in = $3::time
			WHERE attendance.clock_in IS NULL
			RETURNING clock_in
		`, [userId, today, clockInTime]);
		} else {
			res = await client.query(`
			INSERT INTO attendance (user_id, date, clock_in)
			VALUES ($1, $2, CURRENT_TIME)
			ON CONFLICT (user_id, date) DO UPDATE
			SET clock_in = CURRENT_TIME
			WHERE attendance.clock_in IS NULL
			RETURNING clock_in
		`, [userId, today]);
		}

		const recorded = res.rows && res.rows[0] ? res.rows[0].clock_in : null;
		return NextResponse.json({ message: 'Clocked in successfully', clockInTime: recorded }, { status: 200 });
	} catch (error) {
		console.error('Clock-in error:', error.message, error.code, error.stack);
		return NextResponse.json({ 
			error: 'Failed to clock in', 
			details: error.message 
		}, { status: 500 });
	} finally {
		if (client) {
			client.release();
			console.log('Database client released');
		}
	}
}

		// Find today's record
		const existing = await client.query(`
			SELECT clock_in, clock_out FROM attendance
			WHERE user_id = $1 AND date = $2
		`, [userId, today]);

		if (existing.rows[0].clock_out) {
			return NextResponse.json({ error: 'Already clocked out today' }, { status: 400 });

		// Update clock_out

			UPDATE attendance
			SET clock_out = CURRENT_TIME
			WHERE user_id = $1 AND date = $2
			RETURNING clock_out, hours_worked
		`, [userId, today]);

		const { clock_out, hours_worked } = result.rows[0];
		return NextResponse.json({ 
			message: 'Clocked out successfully',
			time: clock_out,
			hoursWorked: hours_worked
		}, { status: 200 });
	} catch (error) {
		console.error('Clock-out error:', error.message, error.code, error.stack);
		return NextResponse.json({ 
			error: 'Failed to clock out', 
			details: error.message 
		}, { status: 500 });
	} finally {
		if (client) client.release();

Integrations & additional explanations
-------------------------------------

1) QR scanner (client-side attendance)
- The staff attendance UI uses a QR scanner to streamline clock-in/clock-out. The scanner is implemented in the staff management attendance page at:
	- `app/dashboard/staff-management/attendance/page.js` (imports `jsqr` and captures camera frames, parses QR payloads).
- QR payload shape expected by the scanner: a small JSON with at least `{ system: "JoannaHotelAttendance", userId: <id> }` or a JWT. The page attempts JSON.parse and falls back to treating the QR as a token for `app/api/auth/qr-login/route.js`.
- When a valid QR (matching `JoannaHotelAttendance`) is scanned the page looks up the `userId` in the loaded users list and calls the appropriate endpoint:
	- Clock-in: `POST /api/attendance/clock-in` (or `POST /api/attendance/clock-in/` depending on client wrapper).
	- Clock-out: `POST /api/attendance/clock-out`.
- Error handling: the client stops the scanner and shows an error message when the QR is invalid, already clocked-in/out, or when camera access fails.

2) Clock-in / clock-out flow summary
- Clock-in path: client sends local time (optional) + userId; server records `clock_in` for today's `date` (en-CA) using `ON CONFLICT` to avoid duplicate entries and returns the recorded time.
- Clock-out path: client requests clock-out by userId; server finds the row for today's date where `clock_in` exists and `clock_out` is null, sets `clock_out = CURRENT_TIME`, and returns `hours_worked` if calculated.
- Timezone notes: the code intentionally records date based on the client's local date string (en-CA) so users near midnight get the intended day. Times are generally recorded with DB server CURRENT_TIME unless the client passes an explicit time string.

3) Housekeeping integration
- Several places in the system raise housekeeping notifications; housekeeping itself has API endpoints to manage tasks and mark notifications read:
	- Notifications insertion points: `app/api/payments/route.js` (inserts housekeeping notifications when a reservation is fully paid), `app/api/room/checkout/route.js` (room checkout triggers housekeeping notification), and `app/api/housekeeping/[id]/route.js` (creates housekeeping tasks and may insert notifications).
- Housekeeping routes:
	- `app/api/housekeeping/[id]/route.js` — get/update housekeeping record for a room, check for existing scheduled notifications, insert notifications when creating tasks, and mark notifications read.
	- `app/api/housekeeping/by-status/route.js` — list housekeeping tasks filtered by status for UI lists.
- Role & UI: Housekeeping users are given the `Housekeeping` role in `app/lib/rbac.js`; the header bell shows for Housekeeping, Manager, Admin. Housekeeping staff typically see notifications targeted at `target_role = 'Housekeeping'` via `GET /api/notifications`.

4) POS and Notifications
- POS (Cafe) payment flows write to `payment_history` and `payments` and may add `notifications` and `audit_logs` depending on the code path:
	- `app/api/payment-history/route.js` is used by POS to persist receipt metadata.
	- `app/api/payments/route.js` inserts to `payments` and — when a reservation reaches full payment — inserts a housekeeping notification.
- Audit & reconciliation: `app/api/audit-logs/cashier-sales/route.js` attempts to reconcile POS receipts with reservations and events for reporting.

5) Notifications architecture (quick reference)
- Table: `notifications` stores messages with columns like `room_id`, `reservation_id`, `message`, `target_role`, `scheduled_time`, `status`, `created_at`.
- API: `app/api/notifications/route.js` supports fetching notifications for a target role and marking notifications read (UPDATE endpoint).
- Service worker: `public/sw.js` contains service worker logic for push events and to display notifications in the browser (vibration, tags). Browser push or client polling can be used to surface new notifications in the UI.

6) Auth / QR login points
- Staff QR generation (for printing or download) exists in the staff list page using `qrcode.react`/`react-qr-code` and produces JSON that the scanner consumes (`app/dashboard/staff-management/staff-list/page.js`).
- QR login endpoint: `app/api/auth/qr-login/route.js` — accepts a scanned QR code payload (JWT or JSON with `qr_code` key) and resolves it to a user session or returns 401 if not recognized.

			);
			setTimeout(() => setSuccessMessage(null), 5000);
			await refreshAttendance();
		} catch (error) {
			console.error("Frontend clock-out error:", error);
			setErrorMessage("Failed to clock out: " + error.message);
			setTimeout(() => setErrorMessage(null), 5000);
		}
	};

	const refreshAttendance = async () => {
		try {
			const res = await fetch("/api/attendance");
			if (!res.ok) throw new Error(`Failed to fetch attendance: ${res.statusText}`);
			const data = await res.json();
			setAttendanceRecords(data);
		} catch (err) {

		// Parse the DB time as a local time (avoid forcing UTC with 'Z') so the

					<button onClick={() => setSuccessMessage(null)} className="text-green-800 font-bold">
						X
					</button>
				</div>
			)}

							<UserGroupIcon className="h-6 w-6" />
						</div>
						<div>
							<p className="text-sm font-medium text-gray-500">Total Users</p>
							<p className="text-2xl font-semibold text-gray-800">{users.length}</p>
						</div>
					</div>
				</div>

															<CheckIcon className="h-4 w-4 text-green-500" />
															<span>{formatTime(record.clock_in)}</span>
														</>
													) : (
														<>
															<XIcon className="h-4 w-4 text-red-500" />
															<span>--:--</span>
														</>
													)}
													<span>/</span>
													{record.clock_out ? (

															<CheckIcon className="h-4 w-4 text-green-500" />
															<span>{formatTime(record.clock_out)}</span>
														</>
													) : (
														<>
															<XIcon className="h-4 w-4 text-red-500" />
															<span>--:--</span>
														</>
													)}
												</div>
											</td>
											<td className="px-6 py-4 whitespace-nowrap text-sm text-gray-500">
												{hoursWorked > 0 ? (
													<span className="flex items-center">
														<ClockIcon className="h-4 w-4 mr-1 text-gray-700" />
														{hoursWorked} hrs
													</span>
												) : (
													"--"
												)}
											</td>
										</tr>
									);
								})
							)}
						</tbody>
					</table>
				</div>
			</div>
		</div>
	);
}
