CREATE TABLE attendance (
    id SERIAL
        PRIMARY KEY,
    user_id INTEGER
        UNIQUE
        NOT NULL,
    date DATE
        UNIQUE
        NOT NULL,
    clock_in TIME,
    clock_out TIME,
    hours_worked DOUBLE PRECISION
        GENERATED ALWAYS AS (
            CASE
                WHEN (clock_in IS NOT NULL) AND (clock_out IS NOT NULL)
                THEN (EXTRACT(epoch FROM (clock_out - clock_in)) / 3600)::numeric
                ELSE NULL::numeric
            END
        ) STORED,
    status VARCHAR(20)
        NOT NULL
        DEFAULT 'present',
    duty_days INTEGER
        DEFAULT 0,
    minutes_late INTEGER
        DEFAULT 0,
    CONSTRAINT attendance_status_check
        CHECK (status::text = ANY (ARRAY['present', 'absent', 'leave']::text[])),
    CONSTRAINT attendance_user_id_fkey
        FOREIGN KEY (user_id)
        REFERENCES public.users(id)
        ON DELETE CASCADE,
    CONSTRAINT attendance_user_id_date_key
        UNIQUE (user_id, date),
    CONSTRAINT attendance_pkey
        PRIMARY KEY (id)
);

CREATE UNIQUE INDEX attendance_pkey
    ON attendance
    USING BTREE (id);

CREATE UNIQUE INDEX attendance_user_id_date_idx
    ON attendance
    USING BTREE (user_id, date);

CREATE UNIQUE INDEX attendance_user_id_date_key
    ON attendance
    USING BTREE (user_id, date);
